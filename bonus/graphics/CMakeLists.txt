function(arcade_extra target folder)
    add_library(${target} MODULE ${ARGN})
    set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden
        VISIBILITY_INLINES_HIDDEN YES PREFIX "" SUFFIX ".so"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
    target_link_libraries(${target} PRIVATE arcade_extra_common)
    set_property(GLOBAL APPEND PROPERTY ARCADE_EXTRA_TARGETS ${target})
    set_property(GLOBAL APPEND PROPERTY ARCADE_EXTRA_FOLDERS ${folder})
endfunction()

function(arcade_pkg target folder package)
    pkg_check_modules(${target}_dep QUIET IMPORTED_TARGET ${package})
    if(${target}_dep_FOUND)
        arcade_extra(${target} ${folder} ${folder}/${ARGN})
        target_link_libraries(${target} PRIVATE PkgConfig::${target}_dep)
        target_link_libraries(arcade_extra_header_dependencies INTERFACE PkgConfig::${target}_dep)
    else()
        message(STATUS "Optional display ${target}: dependency ${package} unavailable")
    endif()
endfunction()
